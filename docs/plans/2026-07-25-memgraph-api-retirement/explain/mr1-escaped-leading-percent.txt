PREPARE mr_search(text,int) AS
SELECT c.id64,c.name
FROM galaxy.system_catalog c
WHERE lower(c.name) LIKE lower($1) || '%' ESCAPE '\'
ORDER BY length(c.name),c.name
LIMIT $2;

EXPLAIN (ANALYZE, BUFFERS, SETTINGS)
EXECUTE mr_search('\%MR0',10);
